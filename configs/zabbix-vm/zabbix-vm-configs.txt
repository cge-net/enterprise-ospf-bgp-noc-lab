template(name="PerHost" type="string" string="/var/log/network/%FROMHOST-IP%.log")

ruleset(name="network") {
    action(type="omfile" dynaFile="PerHost" fileCreateMode="0644")
    action(type="omfile" file="/var/log/network/all.log" fileCreateMode="0644")
}
